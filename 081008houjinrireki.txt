Option Explicit

'===========================================================
' 国税庁 法人番号 Web-API
' 法人番号から変更履歴を取得
'
' Sheet4
'
' C3：法人番号
'
' B5：法人名
' C5：異動年月日
' D5：郵便番号
' E5：県
' F5：市
' G5：番地
' H5：変更内容
' I5：処理区分
' J5：最新
'
' 6行目から出力
'
' 新しい履歴
' ↓
' 古い履歴
'
'===========================================================


'===========================================================
' API設定
'===========================================================

Private Const APP_ID As String = _
    "KjU"

Private Const API_URL As String = _
    "https://api.houjin-bangou.nta.go.jp/4/num"


'===========================================================
' メイン処理
'===========================================================

Public Sub 法人履歴取得()

    Dim ws As Worksheet

    Dim corporateNumber As String

    Dim http As Object
    Dim xml As Object
    Dim corporations As Object

    Dim corp As Object

    Dim url As String

    Dim hitCount As Long
    Dim i As Long
    Dim outputRow As Long

    '-------------------------------------------------------
    ' 対象シート
    '-------------------------------------------------------

    Set ws = ThisWorkbook.Worksheets("Sheet4")


    '-------------------------------------------------------
    ' 法人番号取得
    '-------------------------------------------------------

    corporateNumber = Trim(CStr(ws.Range("C3").value))


    '-------------------------------------------------------
    ' 前回結果を削除
    '-------------------------------------------------------

    ws.Range("B6:J" & ws.rows.count).ClearContents


    '-------------------------------------------------------
    ' 法人番号チェック
    '-------------------------------------------------------

    If corporateNumber = "" Then

        ws.Range("B6").value = "法人番号を入力してください"

        Exit Sub

    End If


    If Not IsNumeric(corporateNumber) Then

        ws.Range("B6").value = "法人番号が不正です"

        Exit Sub

    End If


    If Len(corporateNumber) <> 13 Then

        ws.Range("B6").value = _
            "法人番号は13桁で入力してください"

        Exit Sub

    End If


    '-------------------------------------------------------
    ' 高速化
    '-------------------------------------------------------

    Application.ScreenUpdating = False
    Application.EnableEvents = False

    Application.StatusBar = _
        "法人情報・変更履歴を取得しています..."


    On Error GoTo ErrorHandler


    '=======================================================
    ' API URL
    '
    ' number
    '   法人番号
    '
    ' type=12
    '   XML
    '
    ' history=1
    '   変更履歴を含める
    '
    '=======================================================

    url = API_URL & _
          "?id=" & APP_ID & _
          "&number=" & corporateNumber & _
          "&type=12" & _
          "&history=1"


    '=======================================================
    ' HTTP通信
    '=======================================================

    Set http = CreateObject("MSXML2.XMLHTTP")


    http.Open _
        "GET", _
        url, _
        False


    http.send


    '=======================================================
    ' HTTPステータス確認
    '=======================================================

    If http.Status <> 200 Then

        ws.Range("B6").value = _
            "APIエラー：" & http.Status

        GoTo ExitProcess

    End If


    '=======================================================
    ' XML読み込み
    '=======================================================

    Set xml = _
        CreateObject("MSXML2.DOMDocument.6.0")


    xml.async = False
    xml.validateOnParse = False


    If Not xml.LoadXML(http.responseText) Then

        ws.Range("B6").value = _
            "XML解析エラー"

        GoTo ExitProcess

    End If


    '=======================================================
    ' 法人情報取得
    '=======================================================

    Set corporations = _
        xml.SelectNodes("/corporations/corporation")


    hitCount = corporations.Length


    '=======================================================
    ' 該当なし
    '=======================================================

    If hitCount = 0 Then

        ws.Range("B6").value = _
            "該当法人なし"

        GoTo ExitProcess

    End If


    '=======================================================
    ' 履歴出力
    '
    ' APIから取得した履歴を逆順にする
    '
    ' 最新
    ' ↓
    ' 過去
    '=======================================================

    outputRow = 6


    For i = hitCount - 1 To 0 Step -1

        Set corp = corporations.item(i)


        '===================================================
        ' B列：法人名
        '===================================================

        ws.cells(outputRow, "B").value = _
            GetNodeValue(corp, "name")


        '===================================================
        ' C列：異動年月日
        '===================================================

        ws.cells(outputRow, "C").value = _
            GetNodeValue(corp, "changeDate")


        '===================================================
        ' D列：郵便番号
        '===================================================

        ws.cells(outputRow, "D").NumberFormat = "@"

        ws.cells(outputRow, "D").value = _
            GetNodeValue(corp, "postCode")


        '===================================================
        ' E列：県
        '===================================================

        ws.cells(outputRow, "E").value = _
            GetNodeValue(corp, "prefectureName")


        '===================================================
        ' F列：市
        '===================================================

        ws.cells(outputRow, "F").value = _
            GetNodeValue(corp, "cityName")


        '===================================================
        ' G列：番地
        '===================================================

        ws.cells(outputRow, "G").value = _
            GetNodeValue(corp, "streetNumber")


        '===================================================
        ' H列：変更内容
        '===================================================

        ws.cells(outputRow, "H").value = _
            GetNodeValue(corp, "changeCause")


        '===================================================
        ' I列：処理区分
        '
        ' 数字を日本語に変換
        '===================================================

        ws.cells(outputRow, "I").value = _
            GetProcessName( _
                GetNodeValue(corp, "process"))


        '===================================================
        ' J列：最新
        '===================================================

        If GetNodeValue(corp, "latest") = "1" Then

            ws.cells(outputRow, "J").value = "最新"

        Else

            ws.cells(outputRow, "J").value = ""

        End If


        '===================================================
        ' 次の行
        '===================================================

        outputRow = outputRow + 1


        DoEvents

    Next i


    '=======================================================
    ' 列幅調整
    '
    ' 罫線・色などの書式は変更しない
    '=======================================================

    ws.Columns("B:J").AutoFit


ExitProcess:

    Application.StatusBar = False

    Application.ScreenUpdating = True
    Application.EnableEvents = True

    Exit Sub


'===========================================================
' エラー処理
'===========================================================

ErrorHandler:

    Application.StatusBar = False

    Application.ScreenUpdating = True
    Application.EnableEvents = True


    ws.Range("B6").value = _
        "エラー：" & Err.number & _
        " " & Err.Description

End Sub


'===========================================================
' XMLノードの値を取得
'===========================================================

Private Function GetNodeValue( _
    ByVal parentNode As Object, _
    ByVal nodeName As String) As String


    Dim node As Object


    Set node = _
        parentNode.SelectSingleNode(nodeName)


    If node Is Nothing Then

        GetNodeValue = ""

    Else

        GetNodeValue = node.text

    End If

End Function


'===========================================================
' 処理区分を日本語に変換
'
' 国税庁 法人番号API
'===========================================================

Private Function GetProcessName( _
    ByVal processCode As String) As String


    Select Case processCode


        '---------------------------------------------------
        ' 法人番号指定
        '---------------------------------------------------

        Case "01"

            GetProcessName = _
                "法人番号指定"


        '---------------------------------------------------
        ' 商号・名称変更
        '---------------------------------------------------

        Case "11"

            GetProcessName = _
                "商号・名称変更"


        '---------------------------------------------------
        ' 国内所在地変更
        '---------------------------------------------------

        Case "12"

            GetProcessName = _
                "国内所在地変更"


        '---------------------------------------------------
        ' 国外所在地変更
        '---------------------------------------------------

        Case "13"

            GetProcessName = _
                "国外所在地変更"


        '---------------------------------------------------
        ' 登記記録の閉鎖等
        '---------------------------------------------------

        Case "21"

            GetProcessName = _
                "登記記録の閉鎖等"


        '---------------------------------------------------
        ' 登記記録の復活等
        '---------------------------------------------------

        Case "22"

            GetProcessName = _
                "登記記録の復活等"


        '---------------------------------------------------
        ' 吸収合併
        '---------------------------------------------------

        Case "71"

            GetProcessName = _
                "吸収合併"


        '---------------------------------------------------
        ' 吸収合併無効
        '---------------------------------------------------

        Case "72"

            GetProcessName = _
                "吸収合併無効"


        '---------------------------------------------------
        ' 商号の登記の抹消
        '---------------------------------------------------

        Case "81"

            GetProcessName = _
                "商号の登記の抹消"


        '---------------------------------------------------
        ' 削除
        '---------------------------------------------------

        Case "99"

            GetProcessName = _
                "削除"


        '---------------------------------------------------
        ' その他
        '---------------------------------------------------

        Case Else

            GetProcessName = _
                processCode


    End Select

End Function
